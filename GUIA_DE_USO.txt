4.2. EJEMPLO CON JavaScript (axios)
──────────────────────────────────────────────────────────────────────────────

const axios = require('axios');

const kitsuAPI = axios.create({
  baseURL: 'https://kitsu.io/api/edge',
  headers: {
    'Accept': 'application/vnd.api+json',
    'Content-Type': 'application/vnd.api+json'
  }
});

// Obtener 10 animes
async function get10Animes() {
  const response = await kitsuAPI.get('/anime?page[limit]=10');
  response.data.data.forEach(anime => {
    console.log(anime.attributes.canonicalTitle);
  });
}

// Buscar por categoría
async function getAnimesByCategory(category) {
  const response = await kitsuAPI.get(`/anime?filter[categories]=${category}&page[limit]=10`);
  return response.data.data;
}

// Top 10 más populares
async function getTop10Popular() {
  const response = await kitsuAPI.get('/anime?sort=-userCount&page[limit]=10');
  response.data.data.forEach((anime, index) => {
    console.log(`${index + 1}. ${anime.attributes.canonicalTitle} - ${anime.attributes.userCount} usuarios`);
  });
}

4.3. EJEMPLO CON Python (requests)
──────────────────────────────────────────────────────────────────────────────

import requests

BASE_URL = 'https://kitsu.io/api/edge'
HEADERS = {
    'Accept': 'application/vnd.api+json',
    'Content-Type': 'application/vnd.api+json'
}

# Obtener 10 animes
def get_10_animes():
    response = requests.get(f'{BASE_URL}/anime?page[limit]=10', headers=HEADERS)
    data = response.json()
    for anime in data['data']:
        print(anime['attributes']['canonicalTitle'])

# Buscar por texto
def search_anime(query):
    response = requests.get(f'{BASE_URL}/anime?filter[text]={query}', headers=HEADERS)
    return response.json()['data']

# Top 10 más populares
def get_top_10_popular():
    response = requests.get(f'{BASE_URL}/anime?sort=-userCount&page[limit]=10', headers=HEADERS)
    data = response.json()
    for i, anime in enumerate(data['data'], 1):
        title = anime['attributes']['canonicalTitle']
        users = anime['attributes']['userCount']
        print(f"{i}. {title} - {users} usuarios")

                             7. TIPS Y BUENAS PRÁCTICAS
═══════════════════════════════════════════════════════════════════════════════

✅ Siempre incluye los headers Accept y Content-Type
✅ Usa page[limit] para controlar cuántos resultados obtienes
✅ Combina filtros para búsquedas más específicas
✅ Usa sort para ordenar los resultados
✅ Usa fields para obtener solo los datos que necesitas (ahorra ancho de banda)
✅ Incluye relaciones solo cuando las necesites (include=)
✅ Los nombres de categorías deben estar en minúsculas y con guiones
✅ Codifica URLs correctamente (%20 para espacios, etc.)

❌ No hagas demasiadas peticiones seguidas (respeta rate limits)
❌ No olvides manejar errores (404, 500, etc.)
❌ No asumas que todos los campos estarán presentes (algunos pueden ser null)

                                8. SOLUCIÓN DE PROBLEMAS
═══════════════════════════════════════════════════════════════════════════════

Problema: Error 400 (Bad Request)
Solución: Verifica que los parámetros estén correctamente formateados

Problema: Error 404 (Not Found)
Solución: Verifica que el ID o endpoint exista

Problema: Tests lentos
Solución: Reduce el timeout o limita el número de resultados

Problema: "Cannot find module"
Solución: Ejecuta npm install de nuevo
